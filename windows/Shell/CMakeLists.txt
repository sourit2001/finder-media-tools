cmake_minimum_required(VERSION 3.20)
project(ConvertRightShell LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Build this Explorer extension on Windows with the Windows SDK.")
endif()
add_library(ConvertRightShell SHARED Shell.cpp Shell.def)
target_compile_features(ConvertRightShell PRIVATE cxx_std_17)
target_compile_definitions(ConvertRightShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ConvertRightShell PRIVATE ole32 shell32 shlwapi)
if(MSVC)
  target_compile_options(ConvertRightShell PRIVATE /W4 /WX /permissive- /EHsc)
  set_property(TARGET ConvertRightShell PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
